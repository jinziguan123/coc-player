# 宣传视频实施计划

> **For Claude:** REQUIRED SUB-SKILL: Use superpowers:executing-plans to implement this plan task-by-task.

**Goal:** 按设计稿 `docs/plans/2026-09-25-宣传视频-design.md`，在新目录 `promo/` 搭一条全自动流水线，
产出 B 站 2 分钟宣传片（4K60 MP4）、封面和简介。

**Architecture:** 后端外挂（Python）套在 `app.ai.llm_factory._provider_from_profile` 上，把模型调用
录成磁带、之后回放，骰子换成独立随机源；Playwright 机器人驱动真实前端，回放时录屏并导出时间轴；
Qwen3-TTS 逐句配音；Remotion 按分镜表和时间轴自动剪辑，叠加哈基米、字幕与特效后渲染。
`promo/` 以外的应用代码零改动。

**Tech Stack:** Python 3.10（复用 `server/.venv`：FastAPI、uvicorn、pytest、ruff）· Node 22 + pnpm 11 ·
playwright-core（系统 Chrome）· Remotion 4.0.528（`@remotion/media`）· React 19 · react-icons/gi ·
vitest 4 · mlx-audio 0.5.6（Qwen3-TTS，独立 venv）· ffmpeg 7。

```js
// 对外文案的硬约束：site/README.md「页面文案的硬约束」+ 设计稿 §3.3。
// 叙事在界面上是一段一段出现的（后端按段落切片），所以「逐字」一类说法也算越界。
export const FORBIDDEN = [
  '完整实现', '完整支持', '全部规则', '正式版', '正式发行', '官方授权', '官方合作',
  '逐字', '一个字一个字', '一个词一个词',
]
const EMOJI = /\p{Extended_Pictographic}/gu

export function lintText(text) {
  const issues = []
  for (const word of FORBIDDEN) {
    for (let at = text.indexOf(word); at !== -1; at = text.indexOf(word, at + word.length)) {
      issues.push({ rule: 'forbidden', match: word, index: at })
    }
  }
  for (const m of text.matchAll(EMOJI)) issues.push({ rule: 'emoji', match: m[0], index: m.index })
  return issues.sort((a, b) => a.index - b.index)
}
```

**Step 5：写命令行入口 `promo/scripts/lint-copy.mjs`**

```js
// 扫所有对外文案：旁白稿、Remotion 组件里的字面量、简介。测试文件不扫（里面故意写了违禁词）。
import { existsSync, readdirSync, readFileSync, statSync } from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import { lintText } from './lib/copy-lint.mjs'

const PROMO = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..')
const TARGETS = ['script/narration.yaml', 'src', 'out/description.md']

function* files(rel) {
  const abs = path.join(PROMO, rel)
  if (!existsSync(abs)) return
  if (statSync(abs).isFile()) {
    yield abs
    return
  }
  for (const name of readdirSync(abs)) yield* files(path.join(rel, name))
}

let bad = 0
for (const target of TARGETS) {
  for (const file of files(target)) {
    if (!/\.(ya?ml|tsx?|md)$/.test(file) || /\.test\.tsx?$/.test(file)) continue
    const text = readFileSync(file, 'utf8')
    for (const issue of lintText(text)) {
      const line = text.slice(0, issue.index).split('\n').length
      console.error(`${path.relative(PROMO, file)}:${line}  [${issue.rule}] ${issue.match}`)
      bad++
    }
  }
}
if (bad) {
  console.error(`文案检查未通过：${bad} 处`)
  process.exit(1)
}
console.log('文案检查通过')
```

```python
"""哈基米配音：Qwen3-TTS，经 mlx-audio 在 Apple Silicon 本地运行。

    promo/.venv-tts/bin/python promo/voice/tts.py audition            # 三个候选音色各念三句
    promo/.venv-tts/bin/python promo/voice/tts.py build --engine vivian  # 全部台词 + voice.json

只用模型自带的预置音色，或用文字描述「设计」出来的音色；不克隆任何真人或角色的声音（设计稿 §5）。
首次运行会从 Hugging Face 下载模型（每个 1.7B 模型约 3–4 GB）；下载慢就设
HF_ENDPOINT=https://hf-mirror.com。
"""

from __future__ import annotations

import argparse
import hashlib
import json
import subprocess
from dataclasses import dataclass
from pathlib import Path

import yaml

PROMO = Path(__file__).resolve().parents[1]
NARRATION = PROMO / "script" / "narration.yaml"
AUDITION_DIR = PROMO / ".work" / "voice" / "audition"
VOICE_DIR = PROMO / "public" / "voice"
MANIFEST = PROMO / "public" / "data" / "voice.json"
DESIGN_REF = PROMO / "assets" / "voice" / "design-ref.wav"
DESIGN_REF_TEXT = "你好呀，我是哈基米，今天也要一起跑团喵。"

CUSTOM = "mlx-community/Qwen3-TTS-12Hz-1.7B-CustomVoice-bf16"
DESIGN = "mlx-community/Qwen3-TTS-12Hz-1.7B-VoiceDesign-bf16"
BASE = "mlx-community/Qwen3-TTS-12Hz-0.6B-Base-bf16"

ENGINES = {
    # 预置音色 + 语气指令：音色全片一致，最稳
    "vivian": {"kind": "custom", "model": CUSTOM, "speaker": "Vivian"},
    "serena": {"kind": "custom", "model": CUSTOM, "speaker": "Serena"},
    # 用文字设计一个音色，再以它为参考逐句合成，保证全片是同一个声音
    "design": {
        "kind": "design", "model": DESIGN, "clone_model": BASE,
        "voice": "十七八岁的少女声，声线清亮偏高，语气俏皮，像一只爱撒娇的小猫，尾音常常上扬",
    },
}
AUDITION_LINES = ("hook_1", "s_dice", "t_model")


@dataclass(frozen=True)
class Line:
    id: str
    text: str
    style: str
    meow: bool


def load_lines(path: Path = NARRATION) -> tuple[dict, list[Line]]:
    data = yaml.safe_load(path.read_text(encoding="utf-8"))
    lines = [
        Line(x["id"], x["text"], x.get("style", ""), x.get("meow") == "before")
        for x in data["lines"]
    ]
    ids = [x.id for x in lines]
    if len(ids) != len(set(ids)):
        raise SystemExit("narration.yaml 里有重复的 id")
    return data["voice"], lines


def instruct_for(voice: dict, line: Line) -> str:
    return f"{voice['instruct']}。{line.style}" if line.style else voice["instruct"]


def cache_key(engine: str, instruct: str, text: str) -> str:
    raw = json.dumps([engine, ENGINES.get(engine), instruct, text], ensure_ascii=False)
    return hashlib.sha1(raw.encode("utf-8")).hexdigest()[:16]


def _write(path: Path, results, sample_rate: int) -> None:
    import numpy as np
    import soundfile as sf

    audio = np.concatenate([np.asarray(r.audio, dtype=np.float32) for r in results])
    path.parent.mkdir(parents=True, exist_ok=True)
    sf.write(path, audio, sample_rate)


def _rate(model) -> int:
    return int(getattr(model, "sample_rate", 24000))


def normalize(src: Path, dst: Path) -> None:
    """统一响度（-16 LUFS）、48 kHz 单声道，句与句之间音量才一致。"""
    dst.parent.mkdir(parents=True, exist_ok=True)
    subprocess.run(
        ["ffmpeg", "-y", "-loglevel", "error", "-i", str(src),
         "-af", "loudnorm=I=-16:TP=-1.5:LRA=11", "-ar", "48000", "-ac", "1", str(dst)],
        check=True,
    )


def probe_seconds(path: Path) -> float:
    out = subprocess.run(
        ["ffprobe", "-v", "error", "-show_entries", "format=duration", "-of", "csv=p=0", str(path)],
        check=True, capture_output=True, text=True,
    )
    return round(float(out.stdout.strip()), 3)


class Synth:
    def __init__(self, engine: str, language: str) -> None:
        from mlx_audio.tts.utils import load_model

        self._load = load_model
        self.conf = ENGINES[engine]
        self.language = language
        self.model = load_model(self.conf["model"])
        self.clone = None

    def say(self, text: str, instruct: str, out: Path) -> None:
        if self.conf["kind"] == "custom":
            results = list(self.model.generate_custom_voice(
                text=text, speaker=self.conf["speaker"], language=self.language, instruct=instruct,
            ))
            _write(out, results, _rate(self.model))
            return
        # design：参考音只生成一次并入库（assets/voice/design-ref.wav），之后全片以它为参考
        if not DESIGN_REF.exists():
            ref = list(self.model.generate_voice_design(
                text=DESIGN_REF_TEXT, language=self.language, instruct=self.conf["voice"],
            ))
            _write(DESIGN_REF, ref, _rate(self.model))
        if self.clone is None:
            self.clone = self._load(self.conf["clone_model"])
        results = list(self.clone.generate(
            text=text, ref_audio=str(DESIGN_REF), ref_text=DESIGN_REF_TEXT,
        ))
        _write(out, results, _rate(self.clone))


def cmd_audition(engines: list[str]) -> None:
    voice, lines = load_lines()
    picked = [x for x in lines if x.id in AUDITION_LINES]
    for engine in engines:
        synth = Synth(engine, voice["language"])
        for line in picked:
            raw = AUDITION_DIR / f"{engine}-{line.id}.raw.wav"
            synth.say(line.text, instruct_for(voice, line), raw)
            normalize(raw, AUDITION_DIR / f"{engine}-{line.id}.wav")
            raw.unlink()
    print(f"试音文件在 {AUDITION_DIR}")


def cmd_build(engine: str) -> None:
    voice, lines = load_lines()
    synth: Synth | None = None
    manifest = {"engine": engine, "lines": {}}
    for line in lines:
        instruct = instruct_for(voice, line)
        key = cache_key(engine, instruct, line.text)
        dst = VOICE_DIR / f"{line.id}.wav"
        stamp = VOICE_DIR / f"{line.id}.key"
        if not (dst.exists() and stamp.exists() and stamp.read_text() == key):
            synth = synth or Synth(engine, voice["language"])
            raw = PROMO / ".work" / "voice" / f"{line.id}.raw.wav"
            synth.say(line.text, instruct, raw)
            normalize(raw, dst)
            stamp.write_text(key)
        manifest["lines"][line.id] = {
            "file": f"voice/{line.id}.wav", "seconds": probe_seconds(dst),
            "text": line.text, "meow": line.meow,
        }
    MANIFEST.parent.mkdir(parents=True, exist_ok=True)
    MANIFEST.write_text(json.dumps(manifest, ensure_ascii=False, indent=2), encoding="utf-8")
    total = sum(x["seconds"] for x in manifest["lines"].values())
    print(f"{len(lines)} 句，共 {total:.1f} 秒 → {MANIFEST}")


def main(argv: list[str] | None = None) -> None:
    ap = argparse.ArgumentParser(description="哈基米配音")
    sub = ap.add_subparsers(dest="cmd", required=True)
    a = sub.add_parser("audition")
    a.add_argument("--engines", nargs="+", default=list(ENGINES))
    b = sub.add_parser("build")
    b.add_argument("--engine", required=True, choices=list(ENGINES))
    args = ap.parse_args(argv)
    if args.cmd == "audition":
        cmd_audition(args.engines)
    else:
        cmd_build(args.engine)


if __name__ == "__main__":
    main()
```

**Step 4：写实现 `promo/harness/tape.py`**

```python
"""磁带：把模型调用的输出按通道和先后录下来，之后原样回放。

- 录制（record）：调用透传给真模型，输出逐条追加进 JSONL；进程中途崩了也留得下已录部分。
- 回放（replay）：按「通道 + 先后」取出输出，不再请求模型；流式文本按设定速度吐出。

通道 = 方法名 + 首条 system 消息开头的摘要（先把 UUID、长十六进制串和数字抹平）。同一个调用点
每次的系统提示开头相同，所以后台任务（滚动摘要、幕后推演）和主回合交错执行时各走各的队列，
不会串位；会话 id、房间码、轮次这些每次都不同的值不参与对号。
"""

from __future__ import annotations

import asyncio
import hashlib
import json
import re
from collections import deque
from collections.abc import AsyncIterator, Awaitable, Callable
from pathlib import Path
from typing import Any

HEAD_CHARS = 200
_UUID = re.compile(
    r"[0-9a-fA-F]{8}-?[0-9a-fA-F]{4}-?[0-9a-fA-F]{4}-?[0-9a-fA-F]{4}-?[0-9a-fA-F]{12}"
)
_HEX = re.compile(r"\b[0-9a-fA-F]{16,}\b")
_DIGITS = re.compile(r"\d+")

Sleep = Callable[[float], Awaitable[None]]


class TapeMismatch(RuntimeError):
    """回放对不上：某个通道的录音用完了，或者出现了磁带里没有的通道。需要重录磁带。"""


def _text_of(content: Any) -> str:
    if isinstance(content, str):
        return content
    if isinstance(content, list):  # 多段内容：[{"type": "text", "text": ...}, ...]
        return "".join(p.get("text", "") for p in content if isinstance(p, dict))
    return ""


def normalize_head(text: str) -> str:
    text = _UUID.sub("<id>", text)
    text = _HEX.sub("<hex>", text)
    return _DIGITS.sub("#", text)[:HEAD_CHARS]


def system_head(messages: list[dict]) -> str:
    for m in messages:
        if m.get("role") == "system":
            return _text_of(m.get("content"))
    return _text_of(messages[0].get("content")) if messages else ""


def channel_of(method: str, head_text: str) -> str:
    digest = hashlib.sha1(normalize_head(head_text).encode("utf-8")).hexdigest()[:12]
    return f"{method}:{digest}"


class TapeDeck:
    """一盘磁带。录制时追加写；回放时按通道出队，并负责回放节奏。"""

    def __init__(
        self,
        path: Path | str,
        mode: str,
        *,
        cps: float = 45.0,
        think_seconds: float = 0.4,
        chunk_chars: int = 6,
        sleep: Sleep = asyncio.sleep,
    ) -> None:
        if mode not in ("record", "replay"):
            raise ValueError(f"未知模式：{mode}")
        self.path = Path(path)
        self.mode = mode
        self.cps = cps
        self.think_seconds = think_seconds
        self.chunk_chars = max(1, chunk_chars)
        self._sleep = sleep
        self._seq = 0
        self.header: dict = {}
        self._queues: dict[str, deque[dict]] = {}
        if mode == "replay":
            self._load()

    # ── 录制 ──
    def write_header(self, header: dict) -> None:
        self.path.parent.mkdir(parents=True, exist_ok=True)
        self.header = {"type": "header", **header}
        self.path.write_text(json.dumps(self.header, ensure_ascii=False) + "\n", encoding="utf-8")

    def record(
        self, method: str, head_text: str, result: dict, *,
        model: str | None = None, usage: dict | None = None,
    ) -> None:
        self._seq += 1
        entry = {
            "type": "call",
            "seq": self._seq,
            "channel": channel_of(method, head_text),
            "method": method,
            "head": normalize_head(head_text)[:60],
            "model": model,
            "usage": usage,
            "result": result,
        }
        with self.path.open("a", encoding="utf-8") as f:
            f.write(json.dumps(entry, ensure_ascii=False) + "\n")

    # ── 回放 ──
    def _load(self) -> None:
        for line in self.path.read_text(encoding="utf-8").splitlines():
            if not line.strip():
                continue
            entry = json.loads(line)
            if entry.get("type") == "header":
                self.header = entry
            else:
                self._queues.setdefault(entry["channel"], deque()).append(entry)

    def take(self, method: str, head_text: str) -> dict:
        channel = channel_of(method, head_text)
        queue = self._queues.get(channel)
        if not queue:
            raise TapeMismatch(
                f"磁带里没有更多「{method}」调用可回放（通道 {channel}，系统提示开头："
                f"{normalize_head(head_text)[:60]!r}）。回合编排可能变了，请重录磁带。"
            )
        return queue.popleft()

    def leftover(self) -> dict[str, int]:
        """还没被取走的录音条数。回放结束后不为空，说明这次比录制时少走了几次调用。"""
        return {ch: len(q) for ch, q in self._queues.items() if q}

    async def think(self) -> None:
        if self.think_seconds > 0:
            await self._sleep(self.think_seconds)

    async def paced(self, text: str) -> AsyncIterator[str]:
        step = self.chunk_chars
        delay = step / self.cps if self.cps > 0 else 0.0
        for i in range(0, len(text), step):
            if delay:
                await self._sleep(delay)
            yield text[i:i + step]
```

```python
"""给任意 LLMProvider 套一层磁带：录制时透传并记下输出，回放时直接从磁带出。"""

from __future__ import annotations

from collections.abc import AsyncIterator
from contextlib import aclosing

from app.ai.provider import LLMProvider, StreamDelta, ToolCall
from tape import TapeDeck, system_head

REPLAYED_ERROR = "[磁带回放的异常]"


def _delta_to_dict(d: StreamDelta) -> dict:
    if d.kind == "tool_call" and d.tool_call is not None:
        tc = d.tool_call
        return {"kind": "tool_call", "tool_call": {
            "id": tc.id, "name": tc.name,
            "arguments": tc.arguments, "extra_content": tc.extra_content,
        }}
    return {"kind": d.kind, "text": d.text}


def merge_text(deltas: list[dict]) -> list[dict]:
    """相邻的同类文本增量并成一条：磁带更小，回放时再按速度重新切片。"""
    out: list[dict] = []
    for d in deltas:
        if out and d["kind"] in ("text", "reasoning") and out[-1]["kind"] == d["kind"]:
            out[-1] = {**out[-1], "text": out[-1]["text"] + d["text"]}
        else:
            out.append(dict(d))
    return out


def _error(exc: Exception) -> str:
    return f"{type(exc).__name__}: {exc}"


class TapeProvider(LLMProvider):
    def __init__(self, inner: LLMProvider, deck: TapeDeck) -> None:
        self._inner = inner
        self._deck = deck

    def __getattr__(self, name: str):
        # model 等只读属性透传给真模型（app.ai.provider.model_name() 读的就是 .model）
        if name.startswith("_"):
            raise AttributeError(name)
        return getattr(self._inner, name)

    def supports_tools(self) -> bool:
        return self._inner.supports_tools()

    def supports_vision(self) -> bool:
        return self._inner.supports_vision()

    @property
    def replaying(self) -> bool:
        return self._deck.mode == "replay"

    def _record(self, method: str, head: str, result: dict) -> None:
        self.last_usage = getattr(self._inner, "last_usage", None)
        model = getattr(self._inner, "model", None)
        self._deck.record(
            method, head, result,
            model=model if isinstance(model, str) else None, usage=self.last_usage,
        )

    def _take(self, method: str, head: str) -> dict:
        entry = self._deck.take(method, head)
        self.last_usage = entry.get("usage")
        return entry["result"]

    @staticmethod
    def _raise_if_error(result: dict) -> None:
        if result.get("error"):
            raise RuntimeError(f"{REPLAYED_ERROR} {result['error']}")

    async def complete(
        self, messages, temperature=0.7, max_tokens=None, response_format=None, timeout=None,
    ) -> str:
        head = system_head(messages)
        if self.replaying:
            result = self._take("complete", head)
            await self._deck.think()
            self._raise_if_error(result)
            return result["text"]
        try:
            text = await self._inner.complete(
                messages, temperature=temperature, max_tokens=max_tokens,
                response_format=response_format, timeout=timeout,
            )
        except Exception as exc:
            self._record("complete", head, {"text": "", "error": _error(exc)})
            raise
        self._record("complete", head, {"text": text})
        return text

    async def stream(self, messages, temperature=0.7, max_tokens=None) -> AsyncIterator[str]:
        head = system_head(messages)
        if self.replaying:
            result = self._take("stream", head)
            await self._deck.think()
            async for piece in self._deck.paced(result["text"]):
                yield piece
            self._raise_if_error(result)
            return
        parts: list[str] = []
        error: str | None = None
        try:
            async with aclosing(
                self._inner.stream(messages, temperature=temperature, max_tokens=max_tokens)
            ) as agen:
                async for piece in agen:
                    parts.append(piece)
                    yield piece
        except Exception as exc:
            error = _error(exc)
            raise
        finally:
            result = {"text": "".join(parts)}
            if error:
                result["error"] = error
            self._record("stream", head, result)

    async def stream_chat(
        self, messages, tools=None, temperature=0.7, max_tokens=None,
    ) -> AsyncIterator[StreamDelta]:
        head = system_head(messages)
        if self.replaying:
            result = self._take("stream_chat", head)
            await self._deck.think()
            for d in result.get("deltas", []):
                if d["kind"] == "text":
                    async for piece in self._deck.paced(d["text"]):
                        yield StreamDelta(kind="text", text=piece)
                elif d["kind"] == "reasoning":
                    yield StreamDelta(kind="reasoning", text=d["text"])
                else:
                    tc = d["tool_call"]
                    yield StreamDelta(kind="tool_call", tool_call=ToolCall(
                        id=tc["id"], name=tc["name"],
                        arguments=tc.get("arguments") or {},
                        extra_content=tc.get("extra_content") or {},
                    ))
            self._raise_if_error(result)
            return
        deltas: list[dict] = []
        error: str | None = None
        try:
            async with aclosing(self._inner.stream_chat(
                messages, tools=tools, temperature=temperature, max_tokens=max_tokens,
            )) as agen:
                async for d in agen:
                    deltas.append(_delta_to_dict(d))
                    yield d
        except Exception as exc:
            error = _error(exc)
            raise
        finally:
            result = {"deltas": merge_text(deltas)}
            if error:
                result["error"] = error
            self._record("stream_chat", head, result)

    async def complete_vision(self, prompt, images, max_tokens=None, timeout=None) -> str:
        if self.replaying:
            result = self._take("complete_vision", prompt)
            await self._deck.think()
            self._raise_if_error(result)
            return result["text"]
        try:
            text = await self._inner.complete_vision(
                prompt, images, max_tokens=max_tokens, timeout=timeout,
            )
        except Exception as exc:
            self._record("complete_vision", prompt, {"text": "", "error": _error(exc)})
            raise
        self._record("complete_vision", prompt, {"text": text})
        return text
```

```python
"""宣传片录制用的后端：挂上磁带、给骰子换独立随机源，再以编程方式起 uvicorn。

前端产物（apps/web/dist）由后端同源托管，机器人直接打开 http://127.0.0.1:<port>/。

    server/.venv/bin/python promo/harness/run_backend.py --work promo/.work/session-rec \
        --mode record --tape promo/tapes/session.jsonl --seed 7
    server/.venv/bin/python promo/harness/run_backend.py --work promo/.work/session-play \
        --mode replay --tape promo/tapes/session.jsonl

回放时的种子取自磁带头，命令行的 --seed 只在录制时生效。
"""

from __future__ import annotations

import argparse
import importlib
import json
import os
import random
import subprocess
import sys
from pathlib import Path

from env import ROOT, use_work_dir

# 后端所有掷骰都走这几个模块里的 `random`（server/app/rules/dice.py 等处 `import random`）。
# 统一换成一个独立的 Random 实例：真模型的 HTTP 客户端之类即使也用了全局 random，
# 骰子序列在录制和回放之间也不会错位。
DICE_MODULES = (
    "app.rules.dice",
    "app.rules.coc.checks",
    "app.rules.coc.combat",
    "app.rules.coc.madness",
    "app.services.turn_effects",
)


def ensure_hash_seed() -> None:
    """集合遍历顺序受 PYTHONHASHSEED 影响，必须在解释器启动前定下来，所以不对就自我重启。"""
    if os.environ.get("PYTHONHASHSEED") != "0":
        env = {**os.environ, "PYTHONHASHSEED": "0"}
        os.execve(sys.executable, [sys.executable, *sys.argv], env)


def install_dice(seed: int) -> random.Random:
    rng = random.Random(seed)
    for name in DICE_MODULES:
        importlib.import_module(name).random = rng
    return rng


def install_tape(deck) -> None:
    from app.ai import llm_factory
    from tape_provider import TapeProvider

    original = llm_factory._provider_from_profile
    if getattr(original, "_tape_wrapped", False):
        return

    def wrapped(profile):
        return TapeProvider(original(profile), deck)

    wrapped._tape_wrapped = True
    llm_factory._provider_from_profile = wrapped


def tape_header(work: Path, seed: int) -> dict:
    from seed import public_profile

    settings = json.loads((work / "ai_settings.json").read_text(encoding="utf-8"))
    commit = subprocess.run(
        ["git", "rev-parse", "--short", "HEAD"], cwd=ROOT, capture_output=True, text=True,
    ).stdout.strip()
    return {
        "seed": seed, "commit": commit,
        "profiles": [public_profile(p) for p in settings["profiles"]],
    }


def main(argv: list[str] | None = None) -> None:
    ensure_hash_seed()
    ap = argparse.ArgumentParser(description="宣传片录制用的后端")
    ap.add_argument("--work", required=True)
    ap.add_argument("--mode", choices=("record", "replay"), required=True)
    ap.add_argument("--tape", required=True)
    ap.add_argument("--seed", type=int, default=7)
    ap.add_argument("--port", type=int, default=8011)
    ap.add_argument("--cps", type=float, default=45.0, help="回放：流式文本每秒吐几个字")
    ap.add_argument("--think", type=float, default=0.4, help="回放：每次调用前停几秒")
    ap.add_argument("--dry-run", action="store_true", help="只装好磁带和骰子，打印自检后退出")
    args = ap.parse_args(argv)

    tape_path = Path(args.tape).resolve()
    work = use_work_dir(Path(args.work))
    os.chdir(work)  # 不读任何 .env：录制与回放的配置只来自工作目录

    from tape import TapeDeck

    deck = TapeDeck(tape_path, args.mode, cps=args.cps, think_seconds=args.think)
    if args.mode == "record":
        seed = args.seed
        deck.write_header(tape_header(work, seed))
    else:
        seed = int(deck.header["seed"])
    install_dice(seed)
    install_tape(deck)

    if args.dry_run:
        from app.ai import llm_factory
        from app.rules import dice

        print(json.dumps({
            "mode": args.mode, "seed": seed,
            "dice": [dice.roll_percentile() for _ in range(5)],
            "provider": type(llm_factory.get_llm()).__name__,
        }, ensure_ascii=False))
        return

    import uvicorn
    from app.main import app

    print(f"promo 后端：{args.mode} · 种子 {seed} · http://127.0.0.1:{args.port}", flush=True)
    try:
        uvicorn.run(app, host="127.0.0.1", port=args.port, log_level="warning")
    finally:
        if args.mode == "replay" and deck.leftover():
            print(f"注意：磁带里还有没回放到的调用 {deck.leftover()}", flush=True)


if __name__ == "__main__":
    main()
```

```python
"""把一局导出成 markdown，给作者审故事（审片节点 3）。只读 SQLite，不导入 app。

    server/.venv/bin/python promo/harness/transcript.py --work promo/.work/session-rec \
        --out promo/.work/session-transcript.md
"""

from __future__ import annotations

import argparse
import json
import sqlite3
from pathlib import Path

WHO = {"narration": "KP", "system": "系统", "dice": "骰子", "ooc": "场外"}


def load(db_path: Path) -> tuple[dict, list[dict]]:
    con = sqlite3.connect(db_path)
    con.row_factory = sqlite3.Row
    s = con.execute(
        "select s.id, s.current_scene_id, s.world_state, m.title from game_sessions s "
        "join modules m on m.id = s.module_id order by s.updated_at desc limit 1"
    ).fetchone()
    if s is None:
        raise SystemExit("库里还没有对局")
    rows = con.execute(
        "select sequence_num, event_type, actor_name, content, visibility from event_logs "
        "where session_id = ? order by sequence_num",
        (s["id"],),
    ).fetchall()
    session = {
        "id": s["id"], "title": s["title"], "scene": s["current_scene_id"],
        "world_state": json.loads(s["world_state"] or "{}"),
    }
    return session, [dict(r) | {"visibility": json.loads(r["visibility"] or "[]")} for r in rows]


def render(session: dict, rows: list[dict]) -> str:
    out = [
        f"# 对局记录 · {session['title']}", "",
        f"- 会话：`{session['id']}`",
        f"- 停在场景：`{session['scene']}`",
        f"- 剧情标志：{session['world_state'].get('flags')}", "",
    ]
    for r in rows:
        who = r["actor_name"] or WHO.get(r["event_type"], r["event_type"])
        tag = "（仅 KP 可见）" if r["visibility"] else ""
        out += [f"**#{r['sequence_num']} {who}**{tag}", "", (r["content"] or "").strip(), ""]
    return "\n".join(out)


def main(argv: list[str] | None = None) -> None:
    ap = argparse.ArgumentParser(description="导出对局记录")
    ap.add_argument("--work", required=True)
    ap.add_argument("--out", required=True)
    args = ap.parse_args(argv)
    session, rows = load(Path(args.work) / "trpg.db")
    Path(args.out).write_text(render(session, rows), encoding="utf-8")
    print(f"已导出 {len(rows)} 条事件 → {args.out}")


if __name__ == "__main__":
    main()
```

```js
export function run(cmd, args, opts = {}) {
  return new Promise((resolve, reject) => {
    const p = spawn(cmd, args, { stdio: 'inherit', ...opts })
    p.on('error', reject)
    p.on('exit', (code) => (code === 0 ? resolve() : reject(new Error(`${cmd} 退出码 ${code}`))))
  })
}
```

```js
import { run } from './proc.mjs'

const HARNESS = path.join(PROMO, 'harness')

export async function seedWork({ work, token, profile, fastProfile, fromTape, montage = false }) {
  const args = [path.join(HARNESS, 'seed.py'), '--work', work, '--token', token]
  if (fromTape) args.push('--from-tape', fromTape)
  else {
    args.push('--profile', profile)

export async function startBackend({ work, mode, tape, seed = 7, port = PORT, cps = 45, think = 0.4 }) {
  const proc = spawn(PY, [
    path.join(HARNESS, 'run_backend.py'), '--work', work, '--mode', mode, '--tape', tape,
    '--seed', String(seed), '--port', String(port), '--cps', String(cps), '--think', String(think),
  ], { stdio: ['ignore', 'inherit', 'inherit'], env: { ...process.env, PYTHONHASHSEED: '0' } })
  const base = `http://127.0.0.1:${port}`
  await waitHealthy(`${base}/api/health`, proc)
  return {
    base,
    async stop() {
      if (proc.exitCode !== null) return
      proc.kill('SIGINT')
      await once(proc, 'exit')
    },
  }
}

async function waitHealthy(url, proc, timeoutMs = 90_000) {
  const deadline = Date.now() + timeoutMs
  while (Date.now() < deadline) {
    if (proc.exitCode !== null) throw new Error(`后端提前退出（退出码 ${proc.exitCode}）`)
    try {
      if ((await fetch(url)).ok) return
    } catch { /* 还没起来 */ }
    await new Promise((r) => setTimeout(r, 500))
  }
  throw new Error('后端 90 秒内没起来')
}
```

```js
// 录屏时间轴：关键事件的时刻（带元素位置框）和每次点击的坐标。
// 坐标是 CSS 像素，也就是 1920×1080 视口坐标系，和 Remotion 合成的坐标系一致。
export class Timeline {
  constructor(clock = () => Date.now() / 1000) {
    this.clock = clock
    this.scope = 'x'
    this.markers = []
    this.clicks = []
    this.counters = new Map()
  }

  next(kind) {
    const key = `${this.scope}.${kind}`
    const n = (this.counters.get(key) ?? 0) + 1
    this.counters.set(key, n)
    return `${key}.${n}`
  }

  /** target：CSS 选择器字符串或 Playwright Locator；不给就只记时刻。 */
  async mark(page, name, target) {
    const marker = { name, t: this.clock() }
    if (target) {
      const loc = typeof target === 'string' ? page.locator(target).first() : target
      const box = await loc.boundingBox().catch(() => null)
      if (box) marker.box = box
    }
    this.markers.push(marker)
    return marker
  }

  async click(page, locator, name) {
    const box = await locator.boundingBox().catch(() => null)
    if (box) this.clicks.push({ name, t: this.clock(), x: box.x + box.width / 2, y: box.y + box.height / 2 })
    await locator.click()
  }

  toJSON({ video, t0, duration }) {
    const rel = (t) => Math.round((t - t0) * 1000) / 1000
    return {
      video,
      duration,
      markers: this.markers.map((m) => ({ ...m, t: rel(m.t) })),
      clicks: this.clicks.map((c) => ({ ...c, t: rel(c.t) })),
    }
  }
}
```

```js
// 录屏。两种实现对外接口相同：startCapture(page, opts) → { stop() → { file, t0, duration } }。
// t0 是录屏第一帧的挂钟时间（秒），时间轴里的标记都换算成相对它的秒数。
import { mkdir, open, rm, writeFile } from 'node:fs/promises'
import path from 'node:path'
import { run } from './proc.mjs'

export const DEFAULT_METHOD = 'screencast' // 技术验证后按结论改

const ENCODE = ['-vf', 'fps=60,scale=3840:2160:flags=lanczos,format=yuv420p',
  '-c:v', 'libx264', '-preset', 'slow', '-crf', '14', '-g', '60', '-movflags', '+faststart']

export async function startCapture(page, { method = DEFAULT_METHOD, outFile, workDir }) {
  await mkdir(path.dirname(outFile), { recursive: true })
  if (method === 'tabcapture') return startTabCapture(page, outFile)
  return startScreencast(page, outFile, workDir)
}

async function startScreencast(page, outFile, workDir) {
  await rm(workDir, { recursive: true, force: true })
  await mkdir(workDir, { recursive: true })
  const cdp = await page.context().newCDPSession(page)
  const frames = []
  const writes = []
  cdp.on('Page.screencastFrame', ({ data, metadata, sessionId }) => {
    cdp.send('Page.screencastFrameAck', { sessionId }).catch(() => {})
    const file = path.join(workDir, `f${String(frames.length).padStart(6, '0')}.jpg`)
    frames.push({ file, ts: metadata.timestamp })
    writes.push(writeFile(file, Buffer.from(data, 'base64')))
  })
  await cdp.send('Page.startScreencast', {
    format: 'jpeg', quality: 90, maxWidth: 3840, maxHeight: 2160, everyNthFrame: 1,
  })
  return {
    async stop() {
      const end = Date.now() / 1000
      await cdp.send('Page.stopScreencast').catch(() => {})
      await Promise.all(writes)
      if (!frames.length) throw new Error('一帧都没录到')
      // 画面静止时 screencast 不出帧：补一个结尾帧，让视频一直持续到停止录制那一刻
      frames.push({ file: frames.at(-1).file, ts: Math.max(end, frames.at(-1).ts) })
      const t0 = frames[0].ts
      const list = `${outFile}.frames.txt`
      const lines = []
      frames.forEach((f, i) => {
        const next = frames[i + 1]
        lines.push(`file '${f.file}'`, `duration ${(next ? next.ts - f.ts : 1 / 60).toFixed(4)}`)
      })
      lines.push(`file '${frames.at(-1).file}'`) // concat 的最后一帧要重复一次，时长才生效
      await writeFile(list, lines.join('\n'))
      await run('ffmpeg', ['-y', '-loglevel', 'error', '-f', 'concat', '-safe', '0', '-i', list, ...ENCODE, outFile])
      await rm(list, { force: true })
      await rm(workDir, { recursive: true, force: true })
      return { file: outFile, t0, duration: frames.at(-1).ts - t0, frames: frames.length }
    },
  }
}

// 需要 Chrome 启动参数 --auto-accept-this-tab-capture（见 probe-capture.mjs）
async function startTabCapture(page, outFile, { fps = 60, bitrate = 40_000_000 } = {}) {
  const raw = `${outFile}.raw`
  const fh = await open(raw, 'w')
  let chain = Promise.resolve()
  await page.exposeBinding('__promoChunk', (_src, b64) => {
    const buf = Buffer.from(b64, 'base64')
    chain = chain.then(() => fh.write(buf))
  })
  const cdp = await page.context().newCDPSession(page)
  const starter = async (fps, bitrate) => {
    const stream = await navigator.mediaDevices.getDisplayMedia({
      video: { frameRate: fps }, audio: false, preferCurrentTab: true, selfBrowserSurface: 'include',
    })
    const mimeType = ['video/mp4;codecs=avc1.640034', 'video/webm;codecs=vp9']
      .find((m) => MediaRecorder.isTypeSupported(m))
    const rec = new MediaRecorder(stream, { mimeType, videoBitsPerSecond: bitrate })
    rec.ondataavailable = async (e) => {
      const bytes = new Uint8Array(await e.data.arrayBuffer())
      let s = ''
      for (let i = 0; i < bytes.length; i += 0x8000) s += String.fromCharCode(...bytes.subarray(i, i + 0x8000))
      await window.__promoChunk(btoa(s))
    }
    window.__promoRec = rec
    rec.start(1000)
    return { mimeType, settings: stream.getVideoTracks()[0].getSettings(), t0: Date.now() / 1000 }
  }
  // getDisplayMedia 需要用户手势：走 CDP 的 Runtime.evaluate(userGesture: true)
  const { result, exceptionDetails } = await cdp.send('Runtime.evaluate', {
    expression: `(${starter})(${fps}, ${bitrate})`,
    userGesture: true, awaitPromise: true, returnByValue: true,
  })
  if (exceptionDetails) {
    throw new Error(`标签页采集没起来：${exceptionDetails.exception?.description ?? exceptionDetails.text}`)
  }
  const { t0, settings } = result.value
  return {
    async stop() {
      const end = Date.now() / 1000
      await page.evaluate(() => new Promise((resolve) => {
        window.__promoRec.onstop = () => setTimeout(resolve, 1500)
        window.__promoRec.stop()
      }))
      await chain
      await fh.close()
      await run('ffmpeg', ['-y', '-loglevel', 'error', '-i', raw, ...ENCODE, outFile])
      await rm(raw, { force: true })
      return { file: outFile, t0, duration: end - t0, settings }
    },
  }
}
```

```js
const STABLE_MS = 1500

export async function waitIdle(page, { timeout = 900_000, onFirstNarration } = {}) {
  const deadline = Date.now() + timeout
  const before = await page.locator('.chat-msg--narration').count()
  let since = 0
  while (Date.now() < deadline) {
    const s = await page.evaluate(() => ({
      loading: !!document.querySelector('.chat-loading'),
      advance: !!document.querySelector('[data-tour=advance]'),
      streaming: !!document.querySelector('[data-mid^="stream-"]'),
      narr: document.querySelectorAll('.chat-msg--narration').length,
    }))
    if (onFirstNarration && s.narr > before) {
      await onFirstNarration()
      onFirstNarration = null
    }
    if (!s.loading && s.advance && !s.streaming) {
      since ||= Date.now()
      if (Date.now() - since >= STABLE_MS) return
    } else since = 0
    await page.waitForTimeout(250)
  }
  throw new Error('等 KP 写完超时')
}

export async function fight(page, tl, actions, beats) {
  if (!(await page.locator('.combat-hud').count())) return false
  const id = tl.next('combat')
  await tl.mark(page, `${id}.start`, '.combat-hud')
  beats.push({ kind: 'combat_start' })
  for (let k = 1; k <= 80; k++) {
    await waitCombatDecision(page)
    if (!(await page.locator('.combat-hud').count())) break
    const urgent = page.locator('.combat-zone--urgent')
    if (await urgent.count()) {
      await tl.mark(page, `${id}.defend${k}`, urgent)
      await tl.click(page, urgent.getByRole('button', { name: '闪避', exact: true }), 'dodge')
      actions.push({ kind: 'dodge' })
      continue
    }
    const mine = page.locator('.combat-zone--mine')
    if (await mine.count()) {
      await tl.mark(page, `${id}.turn${k}`, mine)
      await myCombatTurn(page, tl, actions)
    }
  }
  await waitIdle(page) // 战后余波叙事
  await tl.mark(page, `${id}.end`, LAST_NARRATION)
  beats.push({ kind: 'combat_end' })
  return true
}

export async function chase(page, tl, actions, beats) {

  if (!(await card.count())) return false
  const id = tl.next('chase')
  await tl.mark(page, `${id}.start`, card.first())

    const flee = page.getByRole('button', { name: '奔逃一轮', exact: true })
    if (!(await visible(flee, 300_000)) || !(await card.count())) break
    if (!(await flee.isEnabled())) {
      await page.waitForTimeout(500)
      continue
    }
    await tl.click(page, flee, 'flee-round')
    actions.push({ kind: 'flee-round' })
    await waitIdle(page)
    await tl.mark(page, `${id}.round${k}`)
  }
  await waitIdle(page)
  const ending = await page.getByText(/追逐结束：/).last().innerText().catch(() => '')
  await tl.mark(page, `${id}.end`, LAST_NARRATION)

  return true
}

/** 把本轮悬而未决的事都处理掉：检定、战斗、追逐。 */
export async function settle(page, tl, actions, beats) {
  for (let i = 0; i < 20; i++) {
    await waitIdle(page)
    if (await rollPending(page, tl, actions, beats)) continue
    if (await fight(page, tl, actions, beats)) continue
    if (await chase(page, tl, actions, beats)) continue
    return
  }
  throw new Error('settle 循环次数过多：可能卡在一个没处理的界面上')
}

    const r = await fetch(`/api/sessions/${id}`, { headers: { 'X-Player-Token': token } })
    return (await r.json()).current_scene_id
  }, sessionId)
}

/** KP 没自己把人带过去时：先找「前往」建议卡，没有就走大地图。两条路都要再推进一回合。 */
export async function ensureScene(page, tl, actions, beats, sessionId, scene) {
  if ((await currentScene(page, sessionId)) === scene.id) return
  const suggest = page.locator('.chat-msg').filter({ hasText: scene.name })
    .locator('button[title^="把「前往」加入本回合暂存"]')
  if (await suggest.count()) {
    await tl.click(page, suggest.last(), 'travel-suggest')
  } else {
    await tl.click(page, page.locator('button[title="大地图：前往已知地点"]'), 'bigmap')
    await tl.click(page, page.getByRole('button', { name: '调查板', exact: true }), 'board')
    await tl.click(page, page.locator('[title^="前往"]').filter({ hasText: scene.name }).first(), 'pick')
    await tl.click(page, page.getByRole('button', { name: /确认前往/ }), 'confirm')
    await page.keyboard.press('Escape')
  }
  actions.push({ kind: 'travel', scene: scene.id })
  const id = tl.next('travel')
  await tl.click(page, page.locator('[data-tour=advance]'), 'advance')
  await tl.mark(page, `${id}.advance`)
  await settle(page, tl, actions, beats)
  if ((await currentScene(page, sessionId)) !== scene.id) throw new Error(`没能走到「${scene.name}」`)
}
```

```js
const { browser, page } = await openApp({ token: TOKEN, headless: !values.headful })
const tl = new Timeline()
const actions = []
const beats = []
const typeDelay = replay ? 55 : 0 // 回放时让打字在画面里看得见
let capture = null
let failed = null
try {
  await page.goto(`${backend.base}/`, { waitUntil: 'domcontentloaded' })
  if (values.capture) {
    capture = await startCapture(page, {
      method: DEFAULT_METHOD,
      outFile: path.join(PROMO, 'public/footage/session.mp4'),
      workDir: path.join(WORK, 'frames'),
    })

  tl.scope = 'a1'
  const sessionId = await startOnboarding(page, tl)

    tl.scope = act.id
    await ensureScene(page, tl, actions, beats, sessionId, act.scene)
    await tl.mark(page, `${act.id}.enter`)
    for (const text of act.turns) {
      await say(page, tl, actions, text, { typeDelay })
      await settle(page, tl, actions, beats)
    }
  }
} catch (err) {
  failed = err
  await page.screenshot({ path: path.join(WORK, 'failure.png') }).catch(() => {})
} finally {
  const cap = capture ? await capture.stop() : null
  await browser.close()
  await backend.stop()
  if (cap) {
    await mkdir(path.join(PROMO, 'public/data'), { recursive: true })

  }
}
if (failed) throw failed

if (replay) {
  const recorded = JSON.parse(await readFile(ACTIONS, 'utf8'))
  const diff = recorded.findIndex((a, i) => JSON.stringify(a) !== JSON.stringify(actions[i]))
  if (diff !== -1 || recorded.length !== actions.length) {
    console.error('回放走岔了：第', diff + 1, '步', { 录制: recorded[diff], 回放: actions[diff] })
    process.exit(1)
  }
  console.log(`回放与录制一致：${actions.length} 步`)
} else {
  await writeFile(ACTIONS, JSON.stringify(actions, null, 2))
  await writeFile(path.join(PROMO, 'tapes/session.beats.json'), JSON.stringify(beats, null, 2))
  await run(PY, [path.join(PROMO, 'harness/transcript.py'), '--work', WORK, '--out', path.join(PROMO, '.work/session-transcript.md')])

  const missing = Object.entries(need).filter(([, ok]) => !ok).map(([k]) => k)
  if (missing.length || errors) {
    console.error(`缺少：${missing.join('、') || '无'}；报错 ${errors} 次。换个种子重录：--seed ${Number(values.seed) + 1}`)
    process.exit(1)
  }
}
```

```js
// 授权台账校验：视频里用到的每个音频文件、每个字体包，都必须登记来源、许可和署名（设计稿 §5）。
const REQUIRED = ['title', 'author', 'source', 'license', 'license_url']

export function checkLedger(ledger, audioFiles, fontPackages) {
  const problems = []
  const entries = ledger.assets ?? []
  const byFile = new Map(entries.filter((e) => e.file).map((e) => [e.file, e]))
  const byPkg = new Map(entries.filter((e) => e.package).map((e) => [e.package, e]))
  const check = (key, e) => {
    for (const f of REQUIRED) if (!e[f]) problems.push(`${key} 缺少 ${f}`)
    if (/^CC BY/.test(e.license ?? '') && !e.attribution) problems.push(`${key} 是 CC BY，缺署名文本`)
  }
  for (const file of audioFiles) {
    const e = byFile.get(file)
    if (!e) problems.push(`${file} 没有登记`)
    else check(file, e)
  }
  for (const pkg of fontPackages) {
    const e = byPkg.get(pkg)
    if (!e) problems.push(`${pkg} 没有登记`)
    else check(pkg, e)
  }
  return problems
}
```

```js
function* walk(dir) {
  if (!existsSync(dir)) return
  for (const name of readdirSync(dir)) {
    const p = path.join(dir, name)
    if (statSync(p).isDirectory()) yield* walk(p)
    else yield path.relative(path.join(PROMO, 'public'), p)

}
```

**Step 7：写 `promo/scripts/fetch-assets.mjs`**

```js
// 按台账下载素材到 public/，并校验 sha256。fetch 为 null 的条目只检查文件在不在。
import { createHash } from 'node:crypto'
import { existsSync, mkdirSync, readFileSync, writeFileSync } from 'node:fs'
import path from 'node:path'
import { fileURLToPath } from 'node:url'
import YAML from 'yaml'

const PROMO = path.resolve(path.dirname(fileURLToPath(import.meta.url)), '..')
const ledger = YAML.parse(readFileSync(path.join(PROMO, 'assets/ledger.yaml'), 'utf8'))
let bad = 0
for (const e of ledger.assets.filter((x) => x.file)) {
  const dst = path.join(PROMO, 'public', e.file)
  if (!existsSync(dst)) {
    if (!e.fetch) {
      console.error(`需要手动下载：${e.title}（${e.source}）→ ${dst}`)
      bad++
      continue
    }
    const res = await fetch(e.fetch)
    if (!res.ok) throw new Error(`${e.fetch} 下载失败：${res.status}`)
    mkdirSync(path.dirname(dst), { recursive: true })
    writeFileSync(dst, Buffer.from(await res.arrayBuffer()))
  }
  const sha = createHash('sha256').update(readFileSync(dst)).digest('hex')
  if (!e.sha256) console.log(`${e.file} sha256: ${sha}（抄回 ledger.yaml 锁定）`)
  else if (e.sha256 !== sha) {
    console.error(`${e.file} 校验不符：${sha}`)
    bad++
  }
}
process.exit(bad ? 1 : 0)
```

```ts
import '@fontsource/zcool-xiaowei/400.css'
import '@fontsource/zcool-kuaile/400.css'
import { useEffect, useState } from 'react'
import { continueRender, delayRender } from 'remotion'

const FAMILIES = ['700 48px "Noto Serif SC"', '400 48px "ZCOOL XiaoWei"', '400 48px "ZCOOL KuaiLe"']

/** 中文字体按 unicode-range 分片加载：把画面里要用的字都先加载好再开始渲染，否则会闪出替代字体。 */
export function useFonts(texts: string[]): void {
  const [handle] = useState(() => delayRender('加载字体'))
  useEffect(() => {
    const all = texts.join('')
    Promise.all(FAMILIES.map((f) => document.fonts.load(f, all)))
      .finally(() => continueRender(handle))
  }, [])
}
```

```ts
import { MEOW_SRC } from './theme'
import type { Inputs, Shot } from './types'

export const FPS = 60
export const W = 1920
export const H = 1080
export const MAX_RATE = 8
const MEOW_LEAD = 30 // 猫叫比台词早 0.5 秒

export type Track = 'cute' | 'horror'
export type MarkerRef = { shot: string; marker: string; offset?: number }
export type At = number | MarkerRef
export type ClipSpec = {
  shot: string; from: string; to?: string; seconds: number
  pad?: [number, number]; focus?: { marker: string; zoom: number }
}
export type SegmentSpec = {
  id: string; seconds: number; music: Track | null; card?: 'end'
  clips: ClipSpec[]; lines: { id: string; at: At }[]
  sfx?: { src: string; at: At; volume?: number }[]
  effects?: { kind: 'sanFail'; at: At; seconds: number }[]
}

export type CameraKey = { frame: number; zoom: number; cx: number; cy: number }
export type ClipPlan = {
  key: string; shot: string; src: string; from: number; durationInFrames: number
  srcIn: number; rate: number; camera: CameraKey[]; clicks: { frame: number; x: number; y: number }[]
}
export type LinePlan = { id: string; text: string; src: string; from: number; durationInFrames: number }
export type MusicPlan = { track: Track; from: number; durationInFrames: number }
export type EffectPlan = { kind: 'sanFail'; from: number; durationInFrames: number }
export type Timeline = {
  fps: number; durationInFrames: number
  segments: { id: string; from: number; durationInFrames: number; card: 'end' | null }[]
  clips: ClipPlan[]; lines: LinePlan[]; sfx: { src: string; from: number; volume: number }[]
  music: MusicPlan[]; effects: EffectPlan[]
}

function shotOf(inputs: Inputs, name: string): Shot {
  const shot = inputs.shots[name]
  if (!shot) throw new Error(`找不到录屏「${name}」，已有：${Object.keys(inputs.shots).join('、') || '（无）'}`)
  return shot
}

function markerOf(shot: Shot, shotName: string, name: string) {
  const m = shot.markers.find((x) => x.name === name)
  if (m) return m
  const prefix = name.split('.').slice(0, 2).join('.')
  const near = shot.markers.map((x) => x.name).filter((n) => n.startsWith(prefix))
  throw new Error(`录屏「${shotName}」里没有标记「${name}」。相近的有：${near.join('、') || '（无）'}`)
}

function layClips(seg: SegmentSpec, inputs: Inputs, start: number, fps: number): ClipPlan[] {
  let local = start
  return seg.clips.map((spec, i) => {
    const where = `分段「${seg.id}」第 ${i + 1} 个镜头`
    const shot = shotOf(inputs, spec.shot)
    const [padIn, padOut] = spec.pad ?? [0, 0]
    const srcIn = Math.max(0, markerOf(shot, spec.shot, spec.from).t - padIn)
    const natural = spec.to ? markerOf(shot, spec.shot, spec.to).t + padOut - srcIn : spec.seconds
    if (natural <= 0) throw new Error(`${where}：${spec.to} 在 ${spec.from} 之前`)
    const rate = Math.max(1, natural / spec.seconds)
    if (rate > MAX_RATE) {
      throw new Error(`${where}有 ${natural.toFixed(1)} 秒，压进 ${spec.seconds} 秒要 ${rate.toFixed(1)} 倍速，超过上限 ${MAX_RATE}`)
    }
    if (srcIn + spec.seconds * rate > shot.duration + 0.05) throw new Error(`${where}超出了录屏「${spec.shot}」的结尾`)
    const durationInFrames = Math.round(spec.seconds * fps)
    const toLocal = (t: number) => Math.round(((t - srcIn) / rate) * fps)
    const wide: CameraKey = { frame: 0, zoom: 1, cx: W / 2, cy: H / 2 }
    let camera = [wide]
    if (spec.focus) {
      const m = markerOf(shot, spec.shot, spec.focus.marker)
      if (!m.box) throw new Error(`标记「${m.name}」没有位置框，不能推镜头`)
      const at = Math.min(Math.max(toLocal(m.t), 0), durationInFrames)
      const lead = Math.min(36, at) // 0.6 秒推进去
      camera = [wide, { ...wide, frame: at - lead },
        { frame: at, zoom: spec.focus.zoom, cx: m.box.x + m.box.width / 2, cy: m.box.y + m.box.height / 2 }]
    }
    const clip: ClipPlan = {
      key: `${seg.id}-${i}`, shot: spec.shot, src: shot.video, from: local, durationInFrames, srcIn, rate, camera,
      clicks: shot.clicks.map((c) => ({ frame: toLocal(c.t), x: c.x, y: c.y }))
        .filter((c) => c.frame >= 0 && c.frame <= durationInFrames),
    }
    local += durationInFrames
    return clip
  })
}

function frameAt(at: At, seg: SegmentSpec, clips: ClipPlan[], inputs: Inputs, segStart: number, fps: number): number {
  if (typeof at === 'number') return segStart + Math.round(at * fps)
  const m = markerOf(shotOf(inputs, at.shot), at.shot, at.marker)
  const clip = clips.find((c) => c.shot === at.shot && m.t >= c.srcIn && m.t <= c.srcIn + (c.durationInFrames / fps) * c.rate + 1e-6)
  if (!clip) throw new Error(`分段「${seg.id}」里没有镜头覆盖标记「${at.marker}」`)
  return clip.from + Math.round(((m.t - clip.srcIn) / clip.rate) * fps) + Math.round((at.offset ?? 0) * fps)
}

export function buildTimeline(board: SegmentSpec[], inputs: Inputs, fps = FPS): Timeline {
  const tl: Timeline = { fps, durationInFrames: 0, segments: [], clips: [], lines: [], sfx: [], music: [], effects: [] }
  let cursor = 0
  for (const seg of board) {
    const segFrames = Math.round(seg.seconds * fps)
    const clips = layClips(seg, inputs, cursor, fps)
    const used = clips.reduce((n, c) => n + c.durationInFrames, 0)
    if (clips.length && Math.abs(used - segFrames) > 1) {
      throw new Error(`分段「${seg.id}」的镜头加起来 ${(used / fps).toFixed(2)} 秒，和分段时长 ${seg.seconds} 秒对不上`)
    }
    tl.clips.push(...clips)
    const at = (a: At) => frameAt(a, seg, clips, inputs, cursor, fps)
    let lastEnd = cursor
    for (const spec of seg.lines) {
      const v = inputs.voice[spec.id]
      if (!v) throw new Error(`旁白「${spec.id}」还没有配音，先跑 tts.py build`)
      const from = at(spec.at)
      const durationInFrames = Math.ceil(v.seconds * fps)
      if (from < lastEnd) throw new Error(`旁白「${spec.id}」和上一句重叠了 ${((lastEnd - from) / fps).toFixed(2)} 秒`)
      if (from + durationInFrames > cursor + segFrames) throw new Error(`旁白「${spec.id}」超出了分段「${seg.id}」的结尾`)
      tl.lines.push({ id: spec.id, text: v.text, src: v.file, from, durationInFrames })
      if (v.meow) tl.sfx.push({ src: MEOW_SRC, from: Math.max(cursor, from - MEOW_LEAD), volume: 0.9 })
      lastEnd = from + durationInFrames
    }
    for (const cue of seg.sfx ?? []) tl.sfx.push({ src: cue.src, from: at(cue.at), volume: cue.volume ?? 0.8 })
    for (const fx of seg.effects ?? []) {
      tl.effects.push({ kind: fx.kind, from: at(fx.at), durationInFrames: Math.round(fx.seconds * fps) })
    }
    if (seg.music) {
      const prev = tl.music.at(-1)
      if (prev && prev.track === seg.music && prev.from + prev.durationInFrames === cursor) prev.durationInFrames += segFrames
      else tl.music.push({ track: seg.music, from: cursor, durationInFrames: segFrames })
    }
    tl.segments.push({ id: seg.id, from: cursor, durationInFrames: segFrames, card: seg.card ?? null })
    cursor += segFrames
  }
  tl.durationInFrames = cursor
  return tl
}

/** 配乐音量：有人说话时压到 35%，前后各 8 帧渐变；每段配乐最后 12 帧淡出。frame 是全片帧号。 */
export function musicVolume(frame: number, plan: MusicPlan, lines: LinePlan[], base: number): number {
  const RAMP = 8
  let duck = 0
  for (const l of lines) {
    const a = l.from - RAMP
    const b = l.from + l.durationInFrames + RAMP
    if (frame < a || frame > b) continue
    duck = Math.max(duck, Math.min(1, Math.min(frame - a, b - frame) / RAMP))
  }
  const tail = Math.min(1, Math.max(0, (plan.from + plan.durationInFrames - frame) / 12))
  return base * (1 - 0.65 * duck) * tail
}

/** 理智失败的画面抖动：特效开始后 24 帧内衰减，用三角函数保证每次渲染结果一致。 */
export function shakeAt(frame: number, effects: EffectPlan[]): { x: number; y: number } {
  const fx = effects.find((e) => frame >= e.from && frame < e.from + 24)
  if (!fx) return { x: 0, y: 0 }
  const k = 1 - (frame - fx.from) / 24
  return { x: Math.sin(frame * 2.3) * 18 * k, y: Math.cos(frame * 3.1) * 12 * k }
}
```

```tsx
import { Easing, interpolate, useCurrentFrame } from 'remotion'
import type { ClipPlan } from '../timeline'

const MOVE = 24 // 光标在两次点击之间移动用 0.4 秒
const RIPPLE = 18

/** 假光标：脚本操作不会移动系统光标，所以按时间轴里的点击坐标画一个。 */
export const Cursor: React.FC<{ clicks: ClipPlan['clicks'] }> = ({ clicks }) => {
  const frame = useCurrentFrame()
  if (!clicks.length || frame < clicks[0].frame - 30) return null
  let x = clicks[0].x
  let y = clicks[0].y
  for (let i = 1; i < clicks.length; i++) {
    const c = clicks[i]
    const p = clicks[i - 1]
    if (frame >= c.frame) {
      x = c.x
      y = c.y
      continue
    }
    const t = interpolate(frame, [c.frame - MOVE, c.frame], [0, 1], {
      extrapolateLeft: 'clamp', extrapolateRight: 'clamp', easing: Easing.inOut(Easing.quad),
    })
    x = p.x + (c.x - p.x) * t
    y = p.y + (c.y - p.y) * t
    break
  }
  const last = [...clicks].reverse().find((c) => c.frame <= frame)
  const ripple = last ? (frame - last.frame) / RIPPLE : 1
  return (
    <>
      {last && ripple < 1 && (
        <div style={{
          position: 'absolute', left: last.x - 28, top: last.y - 28, width: 56, height: 56, borderRadius: 28,
          border: `3px solid ${THEME.accent}`, transform: `scale(${0.4 + ripple})`, opacity: 1 - ripple,
```

```tsx
import { Video } from '@remotion/media'
import { AbsoluteFill, Easing, staticFile, useCurrentFrame } from 'remotion'
import { H, W, type CameraKey, type ClipPlan } from '../timeline'
import { Cursor } from './Cursor'

function cameraAt(keys: CameraKey[], frame: number) {
  if (frame <= keys[0].frame) return keys[0]
  for (let i = 1; i < keys.length; i++) {
    const a = keys[i - 1]
    const b = keys[i]
    if (frame <= b.frame) {
      const p = b.frame === a.frame ? 1 : Easing.inOut(Easing.cubic)((frame - a.frame) / (b.frame - a.frame))
      return { frame, zoom: a.zoom + (b.zoom - a.zoom) * p, cx: a.cx + (b.cx - a.cx) * p, cy: a.cy + (b.cy - a.cy) * p }
    }
  }
  return keys[keys.length - 1]
}

/** 一段录屏：按时间轴裁切、变速（静音，声音全部在后期另铺），推拉镜头，叠上假光标。 */
export const CameraClip: React.FC<{ clip: ClipPlan; fps: number }> = ({ clip, fps }) => {
  const frame = useCurrentFrame()
  const cam = cameraAt(clip.camera, frame)
  // 镜头中心夹在画面内，放大后不露出画面外的黑边
  const halfW = W / 2 / cam.zoom
  const halfH = H / 2 / cam.zoom
  const cx = Math.min(Math.max(cam.cx, halfW), W - halfW)
  const cy = Math.min(Math.max(cam.cy, halfH), H - halfH)
  return (
    <AbsoluteFill style={{ overflow: 'hidden' }}>
      <div style={{
        position: 'absolute', width: W, height: H, transformOrigin: '0 0',
        transform: `scale(${cam.zoom}) translate(${halfW - cx}px, ${halfH - cy}px)`,
      }}>
        <Video
          src={staticFile(clip.src)}
          trimBefore={Math.round(clip.srcIn * fps)}
          playbackRate={clip.rate}
          muted
          objectFit="fill"
          style={{ width: W, height: H }}
        />
        <Cursor clicks={clip.clicks} />
      </div>
    </AbsoluteFill>
  )
}
```

```ts
import { staticFile } from 'remotion'
import { SHOTS } from './storyboard'
import type { Inputs } from './types'

async function json<T>(file: string): Promise<T> {
  const r = await fetch(staticFile(file))
  if (!r.ok) throw new Error(`读不到 public/${file}（${r.status}）：先跑回放录屏和配音`)
  return r.json() as Promise<T>
}

export async function loadInputs(): Promise<Inputs> {
  const voice = await json<{ lines: Inputs['voice'] }>('data/voice.json')
  const shots: Inputs['shots'] = {}
  for (const name of SHOTS) shots[name] = await json(`data/${name}.json`)
  return { shots, voice: voice.lines }
}
```

```js
// 成片检查：分辨率、帧率、时长、有没有音轨，并测量响度。
import { execFileSync, spawnSync } from 'node:child_process'

const file = process.argv[2]
const probe = JSON.parse(execFileSync('ffprobe', ['-v', 'error', '-show_streams', '-show_format', '-of', 'json', file]))
const v = probe.streams.find((s) => s.codec_type === 'video')
const a = probe.streams.find((s) => s.codec_type === 'audio')
const [num, den] = v.r_frame_rate.split('/').map(Number)
const duration = Number(probe.format.duration)
const checks = {
  '分辨率 3840×2160': v.width === 3840 && v.height === 2160,
  '帧率 60': Math.round(num / den) === 60,
  '时长 118–122 秒': duration >= 118 && duration <= 122,
  '有音轨': !!a,
}
const r = spawnSync('ffmpeg', ['-hide_banner', '-i', file, '-af', 'ebur128=framelog=quiet', '-f', 'null', '-'], { encoding: 'utf8' })
const lufs = Number((r.stderr.match(/I:\s+(-?[\d.]+) LUFS/g) ?? []).pop()?.match(/-?[\d.]+/)?.[0])
console.log({ ...checks, 响度LUFS: lufs, 时长: duration.toFixed(2) })
if (Object.values(checks).includes(false)) process.exit(1)
if (!(lufs >= -18 && lufs <= -12)) console.warn('响度偏离 -16 LUFS 较多，考虑调整配乐或配音的音量')
```
